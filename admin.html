Firebase Production Configuration & Rules Guide

প্রোডাকশন ডেপ্লয়মেন্টের জন্য ফায়ারবেস কনসোলের প্রয়োজনীয় সেটিংস ও সিকিউরিটি রুলস:

১. Authentication Enablement:

Firebase Console > Authentication > Sign-in method > Email/Password চালু (Enable) করুন।
এবং Settings > Authorized domains-এ আপনার ডোমেন kineticdentalcare.com এবং localhost যুক্ত আছে কি না যাচাই করুন।

২. Recommended Cloud Firestore Rules:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isClinicAdmin() {
      return request.auth != null && (
        request.auth.token.email in [
          'kineticdentalwebsite@gmail.com'
        ]
      );
    }
    function isPublicBooking() {
      return request.resource.data.keys().hasOnly([
        'name', 'phone', 'treatment', 'date', 'time', 'notes',
        'createdAt', 'status', 'replyStatus', 'replies'
      ]) &&
      request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 100 &&
      request.resource.data.phone is string && request.resource.data.phone.size() >= 8 && request.resource.data.phone.size() <= 20 &&
      request.resource.data.treatment is string && request.resource.data.treatment.size() > 0 && request.resource.data.treatment.size() <= 120 &&
      request.resource.data.date is string && request.resource.data.date.size() <= 32 &&
      request.resource.data.time is string && request.resource.data.time.size() <= 80 &&
      request.resource.data.notes is string && request.resource.data.notes.size() <= 2000 &&
      request.resource.data.status == 'Pending' &&
      request.resource.data.replyStatus == 'pending' &&
      request.resource.data.replies is list && request.resource.data.replies.size() == 0 &&
      request.resource.data.createdAt == request.time;
    }
    // Patients can submit appointment requests; only clinic admins can read/manage them.
    match /appointments/{appointmentId} {
      allow create: if isClinicAdmin() || isPublicBooking();
      allow get: if true;
      allow list, update, delete: if isClinicAdmin();
    }
    // Public pages need read access; all content changes require clinic admin access.
    match /settings/{settingId} {
      allow read: if true;
      allow write: if isClinicAdmin();
    }
  }
}
